    <script>
        // --- Falling Particles Logic ---
        const particlesContainer = document.getElementById('particles-container');
        let currentParticles = ['💧', '😢', '💙']; // Starts with sad/soft particles
        
        function createParticle() {
            const particle = document.createElement('div');
            particle.className = 'particle';
            particle.innerText = currentParticles[Math.floor(Math.random() * currentParticles.length)];
            particle.style.left = Math.random() * 100 + 'vw';
            particle.style.animationDuration = (Math.random() * 4 + 4) + 's';
            particle.style.fontSize = (Math.random() * 1 + 1) + 'rem';
            particlesContainer.appendChild(particle);
            setTimeout(() => particle.remove(), 8000);
        }
        let particleInterval = setInterval(createParticle, 500);

        // --- Game Logic ---
        const noBtn = document.getElementById('noBtn');
        const yesBtn = document.getElementById('yesBtn');
        const btnContainer = document.getElementById('btnContainer');
        const successMessage = document.getElementById('successMessage');
        const bearGif = document.getElementById('bearGif');
        const titleText = document.getElementById('titleText');
        const messageText = document.getElementById('messageText');

        // Make "No" button run away
        function moveNoButton() {
            const containerRect = btnContainer.getBoundingClientRect();
            const btnRect = noBtn.getBoundingClientRect();

            // Calculate safe bounds within the container
            const maxX = containerRect.width - btnRect.width;
            const maxY = containerRect.height - btnRect.height;

            const randomX = Math.max(0, Math.floor(Math.random() * maxX));
            const randomY = Math.max(0, Math.floor(Math.random() * maxY));

            noBtn.style.left = randomX + 'px';
            noBtn.style.top = randomY + 'px';
        }

        noBtn.addEventListener('mouseover', moveNoButton);
        noBtn.addEventListener('touchstart', (e) => {
            e.preventDefault(); 
            moveNoButton();
        });

        // When she clicks Yes!
        yesBtn.addEventListener('click', () => {
            // Change background and title
            document.body.classList.add('forgiven');
            titleText.innerText = "You're the best! ✨";
            
            // Hide old text and buttons, show success
            messageText.style.display = 'none';
            btnContainer.style.display = 'none';
            successMessage.style.display = 'block';
            
            // Swap sad bear for happy hugging bears
            bearGif.src = "https://media.tenor.com/ef39nFY8b4MAAAAi/bubu-dudu-bubu.gif";
            
            // Change falling drops to happy flowers and hearts
            currentParticles = ['🌸', '💖', '✨', '💕'];
            
            // Burst of happy particles
            for(let i = 0; i < 40; i++) {
                setTimeout(createParticle, i * 50);
            }
        });
    </script>
